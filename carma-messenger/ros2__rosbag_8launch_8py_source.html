27# This function is used to generate a command to record a ROS 2 rosbag that excludes topics
28# topics as provided in the appropriate configuration file.
29def record_ros2_rosbag(context: LaunchContext, rosbag2_qos_override_param_file):
30
31 overriding_qos_profiles = context.perform_substitution(rosbag2_qos_override_param_file)
32
33 proc = ExecuteProcess(
34 cmd=['ros2', 'bag', 'record', '-s', 'mcap', '--qos-profile-overrides-path', overriding_qos_profiles, '-o', '/opt/carma/logs/rosbag2_' + str(datetime.now().strftime('%Y-%m-%d_%H%M%S')), '-a'],
35 output='screen',
36 shell='true'
37 )
38
39 return [proc]
