36 bool sds_service::initialize() {
37 if (!streets_service::initialize() ) {
38 SPDLOG_ERROR("Failed to initialize streets service base!");
39 return false;
40 }
41 SPDLOG_DEBUG("Initializing Sensor Data Sharing Service");
42
43 // Read sensor configuration file and get WSG84 location/origin reference frame.
44 const std::string sensor_id = ss::streets_configuration::get_string_config("sensor_id");
45 const std::string ref_proj_string = ss::streets_configuration::get_string_config("reference_proj_string");
46
47 std::unordered_map<std::string, std::string> proj_string_pairs = parse_proj_string(ref_proj_string);
48 lanelet::GPSPoint ref_proj_gps;
49
50 if (proj_string_pairs.count("lat_0")){
51 ref_proj_gps.lat = std::stod(proj_string_pairs["lat_0"]);
52 }
53 else{
54 SPDLOG_WARN("No latitude value found in reference proj string.");
55 }
56 if (proj_string_pairs.count("lon_0")){
57 ref_proj_gps.lon = std::stod(proj_string_pairs["lon_0"]);
58 }
59 else{
60 SPDLOG_WARN("No longitude value found in reference proj string.");
61 }
62
63 const std::string lanelet2_map = streets_service::get_system_config("LANELET2_MAP", "/home/carma-streets/MAP/Intersection.osm");
64 if (!read_lanelet_map(lanelet2_map)){
65 SPDLOG_ERROR("Failed to read lanelet2 map {0} !", lanelet2_map);
66 return false;
67 }
68
69 this->sdsm_reference_point = ref_proj_gps;
70
71 // Initialize SDSM Kafka producer
72 const std::string sdsm_topic = ss::streets_configuration::get_string_config("sdsm_producer_topic");
73 const std::string detection_topic = ss::streets_configuration::get_string_config("detection_consumer_topic");
74 // Get Infrastructure ID for SDSM messages
75 this->_infrastructure_id = streets_service::get_system_config("INFRASTRUCTURE_ID", "");
76
77 this->_record_detection_metrics = ss::streets_configuration::get_boolean_config("record_detection_metrics");
78 if ( this->_record_detection_metrics ) {
79 std::string header =
80 DETECTION_METRICS_HEADER[0] + "," +
81 DETECTION_METRICS_HEADER[1] + "," +
82 DETECTION_METRICS_HEADER[2] + "," +
83 DETECTION_METRICS_HEADER[3];
84 SPDLOG_INFO("Record detection metrics enabled. Recording metrics {0} ", header);
85 this->_detection_metrics_logger = create_daily_metrics_logger("detection_metrics", header);
86 // Initialize detection metrics
87 this->_detection_metrics[DETECTION_METRICS_HEADER[1]] = 0;
88 this->_detection_metrics[DETECTION_METRICS_HEADER[2]] = 0;
89 this->_detection_metrics[DETECTION_METRICS_HEADER[3]] = 0;
90
91
92 } else {
93 SPDLOG_INFO("Record detection metrics disabled.");
94 }
95
96
97 return initialize_kafka_producer(sdsm_topic, sdsm_producer) && initialize_kafka_consumer(detection_topic, detection_consumer);
98 }

125 void sds_service::consume_detections(){
126 if ( !detection_consumer ) {
127 throw std::runtime_error("Detection consumer is null!");
128 }
129 SPDLOG_DEBUG("Attempting to consume detections ...");
130 detection_consumer->subscribe();
131 while ( detection_consumer->is_running() ) {
132 try{
133 const std::string payload = detection_consumer->consume(1000);
134 if (payload.length() != 0)
135 {
136 auto detected_object = streets_utils::messages::detected_objects_msg::from_json(payload);
137 // Get delay of detected object
138 auto delay = static_cast<int64_t>(ss::streets_clock_singleton::time_in_ms()) - static_cast<int64_t>(detected_object._timestamp);
139
140 if ( this->_record_detection_metrics) {
141 _detection_delay_queue.push_back(delay);
142 if (_detection_delay_queue.size() > 50) {
143 _detection_delay_queue.pop_front();
144 }
145 calculate_detection_delay(_detection_delay_queue, _detection_metrics, DETECTION_METRICS_HEADER[2], _detection_metrics_lock);
146 }
147 // if delay is greater than 500 ms skip detection to get more recent data
148 if ( delay >= 500 ) {
149 SPDLOG_WARN("Skipping incoming detection at {0}ms is not current or has invalid timestamp of {1}ms!" , ss::streets_clock_singleton::time_in_ms(), detected_object._timestamp );
150 if (this->_record_detection_metrics) {
151 increment_drop_metric(_detection_metrics, DETECTION_METRICS_HEADER[1], _detection_metrics_lock);
152 }
153 continue;
154 }
155 // if delay is negative, and service is in simulation mode
156 // the detection message was processed before time sync message. Wait on time sync message
157 else if (is_simulation_mode() && delay < 0 ) {
158 SPDLOG_WARN("Current sim time {0} waiting for sim time {1}ms from detection ...",ss::streets_clock_singleton::time_in_ms(), detected_object._timestamp );
159 ss::streets_clock_singleton::sleep_for(abs(delay));
160 }
161 // If delay is negative and service is not in simulation mode
162 // indicates sensor and service are not time sychronized.
163 else if( delay < 0 ) {
164 SPDLOG_WARN(
165 R"(Skipping incoming detection. Current time is {0}ms and detection time stamp is {1}ms.
166 Sensor Data Sharing Service and sensor producing detections to not appear to be time synchronized.)",
167 ss::streets_clock_singleton::time_in_ms(),
168 detected_object._timestamp );
169 if (this->_record_detection_metrics) {
170 increment_drop_metric(_detection_metrics, DETECTION_METRICS_HEADER[1], _detection_metrics_lock);
171 }
172 continue;
173
174 }
175
176 // Write Lock
177 std::unique_lock lock(detected_objects_lock);
178 detected_objects[detected_object._object_id] = detected_object;
179 SPDLOG_DEBUG("Detected Object List Size {0} after consumed: {1}", detected_objects.size(), payload);
180
181 }
182 }
183 catch (const streets_utils::json_utils::json_parse_exception &e) {
184 SPDLOG_ERROR("Exception occurred consuming detection message : {0}", e.what());
185 if (this->_record_detection_metrics) {
186 increment_drop_metric(_detection_metrics, DETECTION_METRICS_HEADER[1], _detection_metrics_lock);
187 }
188 }
189 }
190 SPDLOG_ERROR("Something went wrong, no longer consuming detections." );
191 }

194 void sds_service::produce_sdsms() {
195 if ( !sdsm_producer ) {
196 throw std::runtime_error("SDSM consumer is null!");
197 }
198 SPDLOG_INFO("Starting SDSM Producer!");
199 unsigned int _interation =0;
200 while ( sdsm_producer->is_running() ) {
201 try{
202 if (this->_record_detection_metrics && _interation % 10 == 0) {
203 write_detection_metrics(_detection_metrics_logger, _detection_metrics, DETECTION_METRICS_HEADER, _detection_metrics_lock);
204 }
205 if ( !detected_objects.empty() ) {
206 streets_utils::messages::sdsm::sensor_data_sharing_msg msg = create_sdsm();
207 const std::string json_msg = streets_utils::messages::sdsm::to_json(msg);
208 SPDLOG_DEBUG("Sending SDSM : {0}", json_msg);
209 sdsm_producer->send(json_msg);
210 // Message Count max is 127, reset after max value
211 if ( this->_message_count < 127) {
212 this->_message_count++;
213 }else {
214 this->_message_count = 0;
215 }
216 // Write Lock
217 std::unique_lock lock(detected_objects_lock);
218 // Clear detected object
219 detected_objects.clear();
220 }
221 }
222 catch( const streets_utils::json_utils::json_parse_exception &e) {
223 SPDLOG_ERROR("Exception occurred producing SDSM : {0}", e.what());
224 if (this->_record_detection_metrics) {
225 increment_drop_metric(_detection_metrics, DETECTION_METRICS_HEADER[3], _detection_metrics_lock);
226 }
227 }
228 _interation++;
229 ss::streets_clock_singleton::sleep_for(100); // Sleep for 100 ms between publish
230 }
231 SPDLOG_CRITICAL("SDSM Producers no longer running.");
232
233 }

235 streets_utils::messages::sdsm::sensor_data_sharing_msg sds_service::create_sdsm() {
236 const std::string ref_proj_string = ss::streets_configuration::get_string_config("reference_proj_string");
237 streets_utils::messages::sdsm::sensor_data_sharing_msg msg;
238 // Read lock
239 uint64_t timestamp = ss::streets_clock_singleton::time_in_ms();
240 msg._time_stamp = to_sdsm_timestamp(timestamp);
241 // Populate with rolling counter
242 msg._msg_count = this->_message_count;
243 // Populate with infrastructure id
244 msg._source_id = this->_infrastructure_id;
245 // Populate equipment type
246 msg._equipment_type = sdsm::equipment_type::RSU;
247 // Populate ref position
248 msg._ref_positon = to_position_3d(this->sdsm_reference_point);
249 std::shared_lock lock(detected_objects_lock);
250 for (const auto &[object_id, object] : detected_objects){
251 try {
252 auto transformed_object_data = detected_object_local_to_ref(object, ref_proj_string);
253 auto ned_object = detected_object_enu_to_ned(transformed_object_data);
254 auto detected_object_data = to_detected_object_data(ned_object, timestamp);
255 // TODO: Update time offset. Currently CARMA-Streets detected object message does not support timestamp
256 // This is a bug and needs to be addressed.
257 msg._objects.push_back(detected_object_data);
258 }
259 catch(const std::exception &e) {
260 SPDLOG_ERROR("Exception: Failed to configure detected object data.", e.what());
261 continue;
262 }
263
264 }
265 return msg;
266 }

278 streets_utils::messages::sdsm::position_3d to_position_3d(const lanelet::GPSPoint &ref_position) {
279 streets_utils::messages::sdsm::position_3d position;
280 // Convert to 1/10 of microdegrees
281 position._longitude = static_cast<int>(ref_position.lon * 1e7);
282 position._latitude = static_cast<int>(ref_position.lat *1e7);
283 // Convert 0.1 meters
284 position._elevation = static_cast<int>(ref_position.ele * 10);
285 return position;

289 void calculate_detection_delay(const std::deque<double> &detection_delay_queue, std::map<std::string, double> &detection_metrics, const std::string &metrics_name, std::mutex &detection_metrics_lock) {
290 SPDLOG_DEBUG("Calculating detection delay for metric: {0}", metrics_name);
291 std::scoped_lock lock{detection_metrics_lock};
292 if (!detection_delay_queue.empty()) {
293 double sum = std::accumulate(detection_delay_queue.begin(), detection_delay_queue.end(), 0.0);
294 detection_metrics[metrics_name] = sum / detection_delay_queue.size();
295 }
296 else {
297 detection_metrics[metrics_name] = 0.0; // No detections, set to 0
298 }
299 SPDLOG_DEBUG("Detection delay calculated: {0} ms", detection_metrics[metrics_name]);
300 }

302 void increment_drop_metric( std::map<std::string, double> &detection_metrics, const std::string &metrics_name, std::mutex &detection_metrics_lock) {
303 std::scoped_lock lock{detection_metrics_lock};
304 detection_metrics[metrics_name] = detection_metrics[metrics_name] + 1;
305 SPDLOG_TRACE("Incrementing drop metric: {0} to {1}", metrics_name, detection_metrics[metrics_name]);
306 }

311 // header includes timestamp so metrics size should be header size - 1
312 if (logger && detection_metrics.size() == metrics_header.size() - 1)
313 {
314 std::scoped_lock lock{detection_metrics_lock};
315 logger->info("{0}, {1}, {2}, {3}",
316 ss::streets_clock_singleton::time_in_ms(),
317 detection_metrics.at(metrics_header[1]),
318 detection_metrics.at(metrics_header[2]),
319 detection_metrics.at(metrics_header[3]));
320 }
321 else {
322 SPDLOG_WARN("Logger is null or detection metrics {0} does not match header size {1} - 1", detection_metrics.size(), metrics_header.size());
323 }
