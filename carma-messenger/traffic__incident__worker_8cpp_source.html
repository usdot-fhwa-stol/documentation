66carma_v2x_msgs::msg::MobilityOperation TrafficIncidentWorker::mobilityMessageGenerator(
67 const gps_msgs::msg::GPSFix & start_zone, const gps_msgs::msg::GPSFix & end_zone)
68{
69 carma_v2x_msgs::msg::MobilityOperation traffic_mobility_msg;
70 // Overload the mobilityMessageGenerator function to use the start and end geofence locations
71 // to get the same message. Get the pinpoint location by getting the midpoint of start and end
72 // and approximate the downtrack and uptrack values using the midpoint.
73 gps_msgs::msg::GPSFix midpoint;
74 midpoint.latitude = (start_zone.latitude + end_zone.latitude) / 2.0;
75 midpoint.longitude = (start_zone.longitude + end_zone.longitude) / 2.0;
76 midpoint.altitude = (start_zone.altitude + end_zone.altitude) / 2.0;
77 midpoint.header.stamp = nh_->now();
78
79 // Calculate approximate distances from midpoint to start and end zones
80 // Using simple lat/lon to meters conversion (approximate for small distances)
81 const double DEGREES_TO_METERS_LAT = 111000.0; // ~111 km per degree latitude
82 const double DEGREES_TO_METERS_LON = 111000.0 * cos(midpoint.latitude * M_PI / 180.0); // Adjusted for longitude
83
84 // Calculate distance to start zone (uptrack - behind the midpoint)
85 double lat_diff_start = start_zone.latitude - midpoint.latitude;
86 double lon_diff_start = start_zone.longitude - midpoint.longitude;
87 double dist = sqrt(pow(lat_diff_start * DEGREES_TO_METERS_LAT, 2) +
88 pow(lon_diff_start * DEGREES_TO_METERS_LON, 2));
89
90 // Set the header information
91 traffic_mobility_msg.m_header.timestamp = rclcpp::Time(midpoint.header.stamp).nanoseconds() / 1e6;
92 traffic_mobility_msg.m_header.sender_id = sender_id_;
93
94 // Set the strategy
95 traffic_mobility_msg.strategy = USE_CASE_NAME_;
96
97 // Build the strategy parameters string using the midpoint coordinates and calculated distances
98 traffic_mobility_msg.strategy_params =
99 "lat:" + doubleToString(midpoint.latitude) + "," +
100 "lon:" + doubleToString(midpoint.longitude) + "," +
101 "downtrack:" + anytypeToString(dist) + "," + "uptrack:" + anytypeToString(dist) +
102 "," + "min_gap:" + anytypeToString(min_gap_) + "," +
103 "advisory_speed:" + anytypeToString(advisory_speed_) + "," + "event_reason:" + event_reason_ +
104 "," + "event_type:" + event_type_;
105
106 return traffic_mobility_msg;
107}

158std::optional<gps_msgs::msg::GPSFix> TrafficIncidentWorker::getGeofenceStartLoc() {
159 if (geo_start_loc_msg_.has_value()) {
160 double age = nh_->now().seconds() - rclcpp::Time(geo_start_loc_msg_.value().header.stamp).seconds();
161 RCLCPP_DEBUG_STREAM(rclcpp::get_logger("traffic_incident"), "Now:"
162 << std::to_string(nh_->now().seconds()) << ", start_msg:"
163 << std::to_string(rclcpp::Time(geo_start_loc_msg_.value().header.stamp).seconds())
164 << ", age" << std::to_string(age) << ", timeout: " << geofence_start_end_data_timeout_);
165 if (age < geofence_start_end_data_timeout_) {
166 return geo_start_loc_msg_.value();
167 }
168 }
169 RCLCPP_DEBUG_STREAM(rclcpp::get_logger("traffic_incident"), "Geo start has no value");
170
171 geo_start_loc_msg_.reset(); // Reset the value if it has expired
172 return std::nullopt;
173}
