18 std::unordered_map<std::string, std::string> parse_proj_string(const std::string& proj_str) {
19 std::unordered_map<std::string, std::string> params;
20 std::stringstream ss(proj_str);
21 std::string token;
22
23 while (ss >> token) {
24 if (!token.empty() && token[0] == '+') {
25 token = token.substr(1);
26 }
27
28 size_t eq = token.find('=');
29 if (eq != std::string::npos) {
30 std::string key = token.substr(0, eq);
31 std::string val = token.substr(eq + 1);
32 params[key] = val;
33 } else {
34 params[token] = "";
35 }
36 }
37 return params;
38 }
