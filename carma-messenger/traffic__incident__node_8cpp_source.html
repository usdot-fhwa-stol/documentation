45 get_parameter<double>("up_track", up_track_);

47 get_parameter<double>("geofence_start_end_data_timeout", geofence_start_end_data_timeout_);
48
49 traffic_worker_->setSenderId(sender_id_);
50 traffic_worker_->setDownTrack(down_track_);
51 traffic_worker_->setUpTrack(up_track_);
52 traffic_worker_->setMinGap(min_gap_);
53 traffic_worker_->setEventReason(event_reason_);
54 traffic_worker_->setEventType(event_type_);
55 traffic_worker_->setGeofenceStartEndDataTimeout(geofence_start_end_data_timeout_);
56
57 // Setup pub/sub
58 pinpoint_driver_sub_ = create_subscription<gps_msgs::msg::GPSFix>(

68 10,
69 std::bind(&TrafficIncidentWorker::geofenceStartLocCallback, traffic_worker_.get(), std_ph::_1));

71 geofence_ending_location_sub_ = create_subscription<gps_msgs::msg::GPSFix>(
72 "gps_fix_end_zone",

75
76 // setup services
77 start_broadcast_request_service_server = create_service<carma_msgs::srv::SetTrafficEvent>(
78 "start_broadcasting_traffic_event",
79 std::bind(
80 &TrafficIncidentNode::startTrafficBroadcastCallback,
81 this,
82 std_ph::_1,
83 std_ph::_2,
84 std_ph::_3));
85 stop_broadcast_request_service_server = create_service<std_srvs::srv::Trigger>(

90 std_ph::_1,
91 std_ph::_2,
92 std_ph::_3));
93
94 return CallbackReturn::SUCCESS;
95}

116/*****
117 * Used by UI to start broadcasting traffic event (geofence)
118 * Msg: MobilityOperation
119 * **/
120bool TrafficIncidentNode::startTrafficBroadcastCallback(
121 const std::shared_ptr<rmw_request_id_t>,
122 const carma_msgs::srv::SetTrafficEvent::Request::SharedPtr req,
123 carma_msgs::srv::SetTrafficEvent::Response::SharedPtr resp)
124{
125 // update instance variables with incoming request params
126 traffic_worker_->setMinGap(req->minimum_gap);
127 traffic_worker_->setDownTrack(req->down_track);
128 traffic_worker_->setUpTrack(req->up_track);
129 traffic_worker_->setAdvisorySpeed(req->advisory_speed);
130 should_broadcast_ = true;
131 // return service response true
132 resp->success = true;
133 return true;
134}

136/*****
137 * Used by UI to stop broadcasting traffic event (geofence)
138 * Msg: MobilityOperation
139 * **/
140bool TrafficIncidentNode::stopTrafficBroadcastCallback(
141 const std::shared_ptr<rmw_request_id_t>,
142 const std_srvs::srv::Trigger::Request::SharedPtr req,
143 std_srvs::srv::Trigger::Response::SharedPtr resp)
144{
145 try {
146 // reset instance variables
147 traffic_worker_->setMinGap(0);
148 traffic_worker_->setDownTrack(0);
149 traffic_worker_->setUpTrack(0);
150 traffic_worker_->setAdvisorySpeed(0);
151 resp->success = true;
152 resp->message = "stop broadcasting";
153 should_broadcast_ = false; // set the flag to false to stop broadcasting
154 // return service response true
155 return true;
156
157 } catch (...) {
158 // in case any exception
159 resp->success = false;
160 return false;
161 }

164void TrafficIncidentNode::spin_callback(void)
165{
166 if (!should_broadcast_)
167 {
168 // If not broadcasting, return
169 return;
170 }
171
172 // Publish traffic mobility operation message only if the worker has valid data
173 // which is also set to invalid (zeroes) if stop broadcast service is called
174 if (
175 std::fabs(traffic_worker_->getDownTrack()) < epsilon_ &&
176 std::fabs(traffic_worker_->getUpTrack()) < epsilon_ &&
177 std::fabs(traffic_worker_->getAdvisorySpeed()) < epsilon_)
178 {
179 RCLCPP_WARN_STREAM(
180 get_logger(), "Traffic Incident Worker has no valid data to publish. "
181 << "Downtrack: " << traffic_worker_->getDownTrack()
182 << ", Uptrack: " << traffic_worker_->getUpTrack()
183 << ", Advisory Speed: " << traffic_worker_->getAdvisorySpeed());
184 return;
185 }
186
187 // If start and end location are set, use that info to generate geofences
188 if (traffic_worker_->getGeofenceStartLoc().has_value() &&
189 traffic_worker_->getGeofenceEndLoc().has_value()) {
190 // start constantly broadcasting using start/end geofence locations
191 traffic_mobility_operation_pub_->publish(
192 traffic_worker_->mobilityMessageGenerator(traffic_worker_->getGeofenceStartLoc().value(),
193 traffic_worker_->getGeofenceEndLoc().value()));
194 return;
195 }
196
197 // If else, use TORC Pinpoint to start constantly broadcasting mobilityOperation msg
198 traffic_mobility_operation_pub_->publish(
199 traffic_worker_->mobilityMessageGenerator(traffic_worker_->getPinPoint()));
200
201}
